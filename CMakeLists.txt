cmake_minimum_required(VERSION 3.20)
project(ecnr LANGUAGES C CXX)

# -----------------------------------------------------------------------------
# Phase 0.5 build:
#   Real WebRTC AEC3 + RNNoise wired behind AecChain (multi-rate 16/48 kHz,
#   2-8 mic, post-Beamformer mono into AEC3). System deps: a C++17 compiler,
#   libsndfile, pkg-config, meson + abseil + speexdsp (for the vendored WebRTC
#   build). googletest is fetched at configure time; miniaudio is vendored.
# -----------------------------------------------------------------------------

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
endif()

add_compile_options(-Wall -Wextra -Wpedantic)

include(FetchContent)
find_package(Python3 COMPONENTS Interpreter)

# --- libsndfile -----------------------------------------------------------------
# Prefer pkg-config (works for both Homebrew and most Linux distros).
find_package(PkgConfig REQUIRED)
pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)

# --- speexdsp (for RNNoise's 16<->48 kHz resampling at 16k tier) ----------------
pkg_check_modules(SPEEXDSP REQUIRED IMPORTED_TARGET speexdsp)

# --- webrtc-audio-processing (vendored, built via Meson + ExternalProject) -----
# Produces the imported target `webrtc::apm`. Downstream code (Phase 0.5
# Task 6) will link against this; nothing in the pipeline uses it yet.
include(cmake/BuildWebRTCAPM.cmake)

# --- ecnr_rnnoise (RNNoise compiled inline as a static lib) ---------------------
# RNNoise is small (~10 .c files); we sidestep its autotools and compile it
# directly. Source list mirrors RNNOISE_SOURCES from upstream Makefile.am.
# `rnnoise_data.c` is generated by `vendor/rnnoise/download_model.sh`; that
# step is also run from `scripts/fetch-vendor.sh` after the rnnoise checkout.
# Phase 0.5 Task 7 wires this into AecChain; nothing uses it yet.
set(RNNOISE_LIB_SOURCES
  vendor/rnnoise/src/denoise.c
  vendor/rnnoise/src/rnn.c
  vendor/rnnoise/src/pitch.c
  vendor/rnnoise/src/kiss_fft.c
  vendor/rnnoise/src/celt_lpc.c
  vendor/rnnoise/src/nnet.c
  vendor/rnnoise/src/nnet_default.c
  vendor/rnnoise/src/parse_lpcnet_weights.c
  vendor/rnnoise/src/rnnoise_data.c
  vendor/rnnoise/src/rnnoise_tables.c
)
add_library(ecnr_rnnoise STATIC ${RNNOISE_LIB_SOURCES})
target_include_directories(ecnr_rnnoise PUBLIC
  vendor/rnnoise/include
  vendor/rnnoise/src
)
set_target_properties(ecnr_rnnoise PROPERTIES POSITION_INDEPENDENT_CODE ON)
# Quiet upstream-only warnings without weakening our project warnings.
target_compile_options(ecnr_rnnoise PRIVATE -Wno-unused-function -Wno-sign-compare)
# DISABLE_DEBUG_FLOAT drops the 7 redundant float-weight arrays from
# rnnoise_data.c that are duplicates of the int8 versions also baked in.
# We've patched init_rnnoise() in rnnoise_data.c to pass NULL for the
# float-weight names of those 7 layers (forcing compute_linear to take
# the int8 path), so the float data is dead bytes. -DDISABLE_DEBUG_FLOAT
# tells the upstream-generated file not to define them at all.
#
# Result on aarch64: -11 MB binary (.rodata drops from 14.8 MB to ~4 MB)
# + RNNoise runs the int8 path (sparse_cgemv8x4 — faster + 1/4 memory
# bandwidth vs sgemv on the GRU forward pass; on A55 with the DotProd
# extension this is also where NEON SDOT helps).
#
# Layers preserved as float-only (conv1, dense_out, vad_dense): their
# float weights live OUTSIDE the DISABLE_DEBUG_FLOAT block in
# rnnoise_data.c and are untouched.
target_compile_definitions(ecnr_rnnoise PRIVATE DISABLE_DEBUG_FLOAT)
# U300 tune (spec 2026-07-04 M4): Cortex-A55 implements the ARMv8.2 Dot
# Product instructions, and RNNoise's int8 GEMV (vec_neon.h) keys its
# fastest path on __ARM_FEATURE_DOTPROD. The Poky stand-in SDK tunes for
# cortexa57 (ARMv8.0) which leaves that path dead. Appended after the SDK
# flags so -mcpu wins for this target only — scoped to the RNNoise hot
# lib, not the whole build, so nothing else strays from the SDK tune.
# qemu-user's default 'max' CPU executes SDOT; A57-class silicon would
# not, which is fine: the deployment target is the U300's A55.
if(CMAKE_CROSSCOMPILING AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  target_compile_options(ecnr_rnnoise PRIVATE -mcpu=cortex-a55)
endif()

# --- googletest -----------------------------------------------------------------
FetchContent_Declare(
  googletest
  GIT_REPOSITORY https://github.com/google/googletest.git

  GIT_SHALLOW TRUE

# --- ONNX Runtime (Phase-3 RES hybrid stage, ADR-0014) -------------------------
# Optional: when onnxruntime is available the DTLN RES branch compiles into
# ecnr_pipeline (enabled at runtime only via AecChain::SetResConfig — zero
# overhead otherwise). ORT is packaged everywhere we build (brew bottle on
# macOS, upstream release tarballs for linux x64/aarch64 — point ORT_HOME
# at an unpacked tarball on CI / Yocto).
option(ECNR_ENABLE_RES "Build the DTLN RES hybrid stage (needs onnxruntime)" ON)
set(ECNR_HAVE_ORT FALSE)
if(ECNR_ENABLE_RES)
  find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h
    HINTS "$ENV{ORT_HOME}/include"
          /opt/homebrew/opt/onnxruntime/include/onnxruntime
          /usr/local/opt/onnxruntime/include/onnxruntime
          /usr/include/onnxruntime)
  find_library(ORT_LIBRARY onnxruntime
    HINTS "$ENV{ORT_HOME}/lib"
          /opt/homebrew/opt/onnxruntime/lib
          /usr/local/opt/onnxruntime/lib)
  if(ORT_INCLUDE_DIR AND ORT_LIBRARY)
    set(ECNR_HAVE_ORT TRUE)
    message(STATUS "ecnr: RES stage ON (onnxruntime: ${ORT_LIBRARY})")
  else()
    message(STATUS "ecnr: RES stage OFF (onnxruntime not found; set ORT_HOME)")
  endif()
endif()

# --- ecnr_core (frame, ring buffer) --------------------------------------------
add_library(ecnr_core STATIC
  src/core/ring_buffer.cc
)
target_include_directories(ecnr_core PUBLIC src)

# --- ecnr_pipeline (AEC chain) -------------------------------------------------
add_library(ecnr_pipeline STATIC
  src/pipeline/aec_chain.cc
  src/pipeline/aec3_adapter.cc
  src/pipeline/agc2_adapter.cc
  src/pipeline/beamformer.cc
  src/pipeline/rnnoise_adapter.cc
)
target_link_libraries(ecnr_pipeline PUBLIC ecnr_core)
# webrtc::apm + ecnr_rnnoise + speexdsp are implementation details — only the
# *_adapter.cc files reference those types. Keep PRIVATE so the public
# AecChain header stays clean.
target_link_libraries(ecnr_pipeline PRIVATE webrtc::apm ecnr_rnnoise PkgConfig::SPEEXDSP)
if(ECNR_HAVE_ORT)
  target_sources(ecnr_pipeline PRIVATE src/pipeline/dtln_res_adapter.cc)
  target_compile_definitions(ecnr_pipeline PRIVATE ECNR_HAVE_ORT=1)
  target_include_directories(ecnr_pipeline PRIVATE ${ORT_INCLUDE_DIR})
  target_link_libraries(ecnr_pipeline PRIVATE ${ORT_LIBRARY})
endif()
# aec3_adapter.cc builds a tuned EchoControlFactory around EchoCanceller3,
# whose header is not part of the installed API — include it from the
# vendored source tree (same pinned revision the archive was built from,
# so the duplicated public headers are byte-identical with the install).
target_include_directories(ecnr_pipeline PRIVATE
  vendor/webrtc-audio-processing
  vendor/webrtc-audio-processing/webrtc
)
# Platform defines the vendored headers key on — must mirror what the
# Meson build compiled the archive with (meson.build:77-110) or the
# header-side types diverge from the object code.
if(APPLE)
  target_compile_definitions(ecnr_pipeline PRIVATE WEBRTC_POSIX WEBRTC_MAC)
elseif(UNIX)
  target_compile_definitions(ecnr_pipeline PRIVATE WEBRTC_POSIX WEBRTC_LINUX)
endif()
target_compile_definitions(ecnr_pipeline PRIVATE WEBRTC_APM_DEBUG_DUMP=0)
# The vendored headers are not -Wpedantic-clean under our warning set;
# keep project warnings for our code but drop the nullability noise the
# apple clang extension check adds on scoped_refptr.h.
target_compile_options(ecnr_pipeline PRIVATE -Wno-nullability-extension -Wno-nullability-completeness)

# --- ecnr_hal (file I/O via libsndfile) ----------------------------------------
add_library(ecnr_hal STATIC
  src/hal/file_io.cc
)
target_link_libraries(ecnr_hal PUBLIC ecnr_core PkgConfig::SNDFILE)

# --- ecnr_eval (AEC3 tuning + ERLE measurement harness) -----------------------
# Per ADR-0011. Static lib `ecnr_eval_metrics` holds the pure ERLE math
# (no chain dependency, fully unit-testable); the executable wires it
# into the chain.
add_library(ecnr_eval_metrics STATIC
  src/eval/metrics.cc
)
target_link_libraries(ecnr_eval_metrics PUBLIC ecnr_core)

option(ECNR_BUILD_EVAL "Build the ecnr_eval AEC3 tuning harness" ON)
if(ECNR_BUILD_EVAL)
  add_executable(ecnr_eval src/eval/eval_main.cc)
  target_link_libraries(ecnr_eval PRIVATE
    ecnr_pipeline ecnr_hal ecnr_eval_metrics)
endif()

# --- ecnr_bench (offline harness) ----------------------------------------------
add_executable(ecnr_bench src/bench/main.cc)
target_link_libraries(ecnr_bench PRIVATE ecnr_pipeline ecnr_hal)

# --- ecnr_live (host loopback E2E via miniaudio) -------------------------------
# miniaudio is a single-header library — we vendor the pinned header at
# third_party/miniaudio/. Linker flags below cover macOS CoreAudio; on Linux
# we link ALSA + pthread + dl + m.
#
# Gated behind ECNR_BUILD_LIVE because the host loopback harness is not a
# deployment artefact — it's a macOS / Linux desktop development tool. The
# aarch64 cross-compile spike (scripts/cross-build-yocto/) disables this
# target so we don't drag miniaudio + ALSA into the cross-build before
# we've decided whether ecnr_live runs on the U300 HAL at all.
option(ECNR_BUILD_LIVE "Build the ecnr_live host loopback harness" ON)
if(ECNR_BUILD_LIVE)
  add_executable(ecnr_live src/live/live_loopback.cc)
  target_link_libraries(ecnr_live PRIVATE ecnr_pipeline ecnr_hal)
  target_include_directories(ecnr_live PRIVATE third_party/miniaudio)
  if(APPLE)
    target_link_libraries(ecnr_live PRIVATE
      "-framework CoreFoundation"
      "-framework CoreAudio"
      "-framework AudioToolbox"
      "-framework AudioUnit"
    )
  elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(ecnr_live PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
    # IMPORTED_TARGET is required for PkgConfig::ALSA to be created. The
    # historic no-IMPORTED_TARGET form silently produced ALSA_FOUND but no
    # PkgConfig::ALSA target; the subsequent target_link_libraries() then
    # failed at configure time on any machine with a working ALSA.
    pkg_check_modules(ALSA IMPORTED_TARGET alsa)
    if(ALSA_FOUND)
      target_link_libraries(ecnr_live PRIVATE PkgConfig::ALSA)
    endif()
  endif()
endif()
